    35→// ============ API Helper ============
    36→async function api(path: string, options?: RequestInit) {
    37→  const token = typeof window !== "undefined" ? localStorage.getItem("quiz_token") : null;
    38→  const res = await fetch(path, {
    39→    ...options,
    40→    headers: {
    41→      ...options?.headers,
    42→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    43→      ...(options?.body instanceof FormData ? {} : { "Content-Type": "application/json" }),
    44→    },
    45→  });
    46→  return res.json();
    47→}
